<%*
// ===== 新周记模板：自动创建 site/content/posts/weekly/<年>-w<周>/index.md =====
// 触发方式：在 posts/weekly/ 下新建笔记（自动），或手动插入模板
let originalTitle = tp.file.title;
if (!originalTitle || originalTitle === "无标题" || originalTitle === "Untitled") {
  originalTitle = await tp.system.prompt("周记标题（如：W34 周总结）：");
}
const date = tp.date.now("YYYY-MM-DD");

// ISO 周号计算（纯 JS，不依赖 moment）
const now = new Date();
const target = new Date(now.getTime());
const day = (target.getDay() + 6) % 7;          // 周一 = 0
target.setDate(target.getDate() - day + 3);     // 本周四
const firstThursday = new Date(target.getFullYear(), 0, 4);
const fday = (firstThursday.getDay() + 6) % 7;
firstThursday.setDate(firstThursday.getDate() - fday + 3);
const week = 1 + Math.round((target - firstThursday) / (7 * 24 * 3600 * 1000));
const year = target.getFullYear();
const slug = `${year}-w${week}`;

await tp.file.move(`site/content/posts/weekly/${slug}/index`); // Obsidian 自动补 .md
tR += `---
title: ${originalTitle}
date: ${date}
lastmod: ${date}
issue: ${week}
tags: [weekly]
categories:
  - 周记
draft: false
description: ""
---

> 本周一句话总结

## 本周完成

- [ ] 内容 1

## 卡点 / 思考

- 问题 1

## 下周计划

- [ ] 计划 1

  站内链接：[文字](/posts/文章slug/)
  周记会自动出现在首页周记卡片（ISSUE 编号 = 周号）
-->
`;
%>
